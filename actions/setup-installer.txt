// Setup Installer - Creates all source modules in /Installer folder
// Run this action ONCE to populate the installer with all modules

function setupInstallerModules() {
    // Create /Installer folder if it doesn't exist
    if(!exists("/Installer")) {
        create("/Installer");
        $Badge("/Installer") = "file-cabinet";
    }
    
    // Create Core Utilities module

'// Core Utilities for Tinderbox Workspace
// Functions for logging, validation, and basic operations

// Function: logMessage
// Purpose: Centralized logging to /Log note with timestamp
function logMessage(message) {
   if(!exists("/Log")) {
      create("/Log");
      $Badge("/Log") = "edit";
      $Text("/Log") = "";
   }
   $Text("/Log") = $Text("/Log") + "[" + format(now,"h:mm:ss") + "] " + message + "\n";
}

// Function: setupContainer
// Purpose: Create container with badge and optional additional setup
function setupContainer(vPath, vBadge, vAdditionalSetup) {
    create(vPath);
    $Badge(vPath) = vBadge;
    if(vAdditionalSetup != "") {
        eval(vAdditionalSetup);
    }
}

// Function: addPrefix
// Purpose: Ensures that the note name starts with the specified prefix
function addPrefix(prefix){
   if ($Name.substr(0, prefix.size()) != prefix) { 
      $Name = prefix + $Name;
   }
}

// Function: childFolder
// Purpose: Creates a new child folder under the given root with combined name
function childFolder(vRoot, vFolderName) {
   var(vName, vFolder){
      vName = $Name(vRoot);
      vFolder = create(vRoot, vName + " " + vFolderName);
      $Prototype(vFolder) = "pFolder";
      $Badge(vFolder) = "folder2";
   }
   return vFolder;
}

// Function: ensureLogExists
// Purpose: Creates log note if it does not exist with proper formatting
function ensureLogExists() {
   if(!exists("/Log")) {
      create("/Log");
      $Badge("/Log") = "edit";
      $TextFont("/Log") = "Courier New";
      $TextFontSize("/Log") = 11;
      $Text("/Log") = "=== Tinderbox Workspace Log ===\n";
   }
}

// Function: validatePrerequisites
// Purpose: Check if workspace is ready for initialization
function validatePrerequisites() {
   var(isValid){
      isValid = true;
      
      if($Name == "untitled") {
         logMessage("ERROR: Cannot initialize untitled document");
         isValid = false;
      }
      
      if($IsProcessing == true) {
         logMessage("WARNING: Processing already in progress");
         isValid = false;
      }
   }
   return isValid;
}

// Function: setSiblingOrder
// Purpose: Set ordering for main workspace containers
function setSiblingOrder() {
   var(vRootCount, vContainers, vIndex){
      vRootCount = $ChildCount(parent);
      vContainers = "/Hints;/Prototypes;/Templates;/Projects;/Resources";
      vIndex = 0;
      
      vContainers.split(";").each(vContainer){
         if(exists(vContainer)) {
            $SiblingOrder(vContainer) = vRootCount - vIndex;
            vIndex = vIndex + 1;
         }
      }
   }
}

// Function: createHintsStructure
// Purpose: Creates the /Hints hierarchy for function libraries
function createHintsStructure() {
   logMessage("Creating /Hints structure...");
   
   setupContainer("/Hints", "folder2", "");
   setupContainer("/Hints/Library", "books", "");
   setupContainer("/Hints/Preview", "display-chart", "");
   
   logMessage("Hints structure created successfully");
}

        "td{" +
        "padding: 0.5em;" +
        "}" +
        "img{" +

// Function: createStyleNote
// Purpose: Create CSS style note in Hints/Preview
function createStyleNote() {
   create("/Hints/Preview","style");
   $Text("/Hints/Preview/style") = getPreviewStyles();
   $Searchable("/Hints/Preview/style") = false;
}';

'// Simple logging function (standalone - no dependencies)
function simpleLog(message) {
    if(!exists("/Log")) {
        create("/Log");
        $Badge("/Log") = "edit";
        $TextFont("/Log") = "Courier New";
        $TextFontSize("/Log") = 11;
        $Text("/Log") = "=== Tinderbox Installation Log ===\n";
    }
    $Text("/Log") = $Text("/Log") + "[" + format(now,"h:mm:ss") + "] " + message + "\n";

// Function to copy a library module to Hints/Library
function copyModuleToLibrary(moduleName, sourceNoteName) {
    var(targetPath, sourceText){
        targetPath = "/Hints/Library/" + moduleName;
        
        simpleLog("Attempting to copy: " + sourceNoteName + " -> " + targetPath);
        simpleLog("  Source exists: " + exists(sourceNoteName));
        
        if(exists(sourceNoteName)) {
            sourceText = $Text(sourceNoteName);
            simpleLog("  Source text length: " + sourceText.size());
            
            if(sourceText != "") {
                simpleLog("  Creating library note: " + moduleName);
                create("/Hints/Library", moduleName);
                $Text(targetPath) = sourceText;
                $Badge(targetPath) = "code";
                $Searchable(targetPath) = false;
                
                simpleLog("  Verifying copy - target exists: " + exists(targetPath));
                simpleLog("  Target text length: " + $Text(targetPath).size());
                simpleLog("SUCCESS: Copied " + moduleName + " (" + sourceText.size() + " chars)");
            } else {
                simpleLog("ERROR: " + sourceNoteName + " exists but has no content");
            }
        } else {
            simpleLog("ERROR: Source note not found: " + sourceNoteName);
        }
    }
}

// Function to install all workspace modules
function installWorkspaceModules() {
    // Ensure /Hints/Library exists
    if(!exists("/Hints")) {
        create("/Hints");
        $Badge("/Hints") = "folder2";
    }
    
    if(!exists("/Hints/Library")) {
        create("/Hints/Library");
        $Badge("/Hints/Library") = "books";
    }
    
    simpleLog("Found /Installer folder, copying modules...");
    
    // Debug: List what is in the /Installer folder
    simpleLog("Contents of /Installer: " + collect(children("/Installer"), $Name).format("; "));
    
    // Copy modules to library from /Installer folder
    copyModuleToLibrary("CoreUtils", "/Installer/Core Utilities");
    copyModuleToLibrary("WorkspaceInitializer", "/Installer/Workspace Initializer");
}

            create("/Hints/Library");
            $Badge("/Hints/Library") = "books";
        }

            create("/Hints/Preview");

        // Create project containers
        simpleLog("Phase 2: Creating project containers...");

            create("/Projects");

        if(!exists("/Resources")) {
            create("/Resources");
            $Badge("/Resources") = "folder2";
            simpleLog("Created /Resources folder");
        }
        
        // Final summary
        simpleLog("=== WORKSPACE SUMMARY ===");

        simpleLog("Library modules: " + $ChildCount("/Hints/Library"));
        
        simpleLog("Installation completed successfully at " + format(date("today"),"D b Y"));
    } else {
        simpleLog("ERROR: CoreUtils not found in library - installation failed");
    }
}
