// Helper functions for common values
function getBadge(type) {
    if(type == "FOLDER") return "folder2";
    if(type == "PROJECT") return "address-book";
    if(type == "STORY") return "chat-ellipsis";
    if(type == "ROLE") return "users";
    if(type == "REQUIREMENT") return "calculator-large";
    if(type == "ACTION") return "crosshair";
    if(type == "HTML") return "globe";
    if(type == "TEMPLATE") return "display-chart";
    if(type == "CABINET") return "file-cabinet";
}

function getTemplate(type) {
    if(type == "PREVIEW_FOLDER") return "/Templates/Preview Folder";
    if(type == "PREVIEW_LEAF") return "/Templates/Preview Leaf";
    if(type == "PREVIEW_STORY") return "/Templates/Preview Story";
}

function getPrefix(type) {
    if(type == "FUNCTIONAL_REQ") return "FR-";
    if(type == "NON_FUNCTIONAL_REQ") return "NFR-";
}

// Function: addPrefix
// Purpose: Ensures that the note's name starts with the specified prefix.
//          If it doesn't, the prefix is prepended to the current name.
function addPrefix(prefix){
   if ($Name.substr(0, prefix.length) != prefix) { 
      $Name = prefix + $Name;
   }
}


// Function: childFolder
// Purpose: Creates a new child folder under the given root with a name that combines
//          the root's name and the provided folder name. Also sets an icon badge.
function childFolder(vRoot, vFolderName) {
   var:string vName = $Name(vRoot);
   var:string vFolder = create(vRoot, vName + " " + vFolderName);
   $Prototype(vFolder) = pFolder;
   $Badge(vFolder) = getBadge("FOLDER");
   return vFolder;
}

// Function: buildUserStory
// Purpose: Configures a note to function as a User Story by setting its prototype,
//          badge, displayed attributes, and an on-visit action.
function buildUserStory() {
   $IsPrototype = false;
   $Prototype = pStory;
   $Badge = getBadge("STORY");
   $DisplayedAttributes = "StoryRole;StoryWant;StoryReason;Prototype;HTMLExportTemplate";
   $OnVisit = "updateRoleList();";
}

// Function: updateRoleList

function updateRoleList() {
   attribute("StoryRole")[suggested] = $Name(children(nextSibling(parent)));

// Function: buildRole
// Purpose: Configures a note to function as a Role by setting its prototype, badge,
//          and suggested attribute values.
function buildRole() {
   $IsPrototype = false;
   $Prototype = pRole;
   $DisplayedAttributes = "HTMLExportTemplate";
   attribute("StoryRole")[suggested] = collect(siblings, $Name);
}

// Function: buildFR
// Purpose: Configures a note as a Functional Requirement by setting its prototype,
//          ensuring its name starts with "FR-", and setting an on-visit action.
function buildFR() {
   $IsPrototype = false;
   $Prototype = pFR;
   addPrefix(getPrefix("FUNCTIONAL_REQ"));
   $OnVisit = "addPrefix('" + getPrefix("FUNCTIONAL_REQ") + "');";
   $DisplayedAttributes = "";
}

// Purpose: Configures a note as a Non-Functional Requirement by setting its prototype,

// Helper functions for project setup
function configureProjectPrototype() {
   $IsPrototype = false;
   $Prototype = "pProject";
   $Badge = getBadge("PROJECT");
   $HTMLExportTemplate = getTemplate("PREVIEW_FOLDER");
}

function createProjectFolders() {
   var:string vRoot = $Path;
   
   // Create User Stories folder
   var:string vUserStories = childFolder(vRoot, "User Stories");
   $OnAdd(vUserStories) = "buildUserStory();";

   // Create Roles folder  
   var:string vRoles = childFolder(vRoot, "Roles");
   $OnAdd(vRoles) = "buildRole();";

   // Create Functional Requirements folder
   var:string vFunctionalRequirements = childFolder(vRoot, "Functional Requirements");
   $OnAdd(vFunctionalRequirements) = "buildFR();";

   // Create Non-Functional Requirements folder
   var:string vNonFunctionalRequirements = childFolder(vRoot, "Non-Functional Requirements");
   $OnAdd(vNonFunctionalRequirements) = "buildNFR();";
}

function setupProjectStructure() {
   configureProjectPrototype();
   createProjectFolders();
}

// Function: buildProject
// Purpose: Sets up a Project note by creating standard child folders (like User Stories,
//          Roles, Functional/Non-Functional Requirements, Tests, and Misc) and assigning
//          on-add actions for some folders.
function buildProject() {
   logMessage("buildProject() called for: " + $Name);
   
   // Validate preconditions
   if ($IsProcessing == true || $Name == "untitled") {
      logMessage("buildProject() skipped - processing flag or untitled name");
      return;
   }
   
   $IsProcessing = true;
   logMessage("Setting up project structure...");
   
   // Only proceed if the note does not have a prototype defined
   if ($Prototype == "") {
      setupProjectStructure();
      logMessage("Project structure created successfully for: " + $Name);
   } else {
      logMessage("buildProject() skipped - prototype already defined: " + $Prototype);
   }
   
   // Reset processing flag
   $IsProcessing = false;
}

// Helper function for logging
function logMessage(message) {
   if(!exists("/Log")) {
      create("/Log");
      $Badge("/Log") = "edit";
      $Text("/Log") = "";
   }
   $Text("/Log") = $Text("/Log") + "[" + format(now,"h:mm:ss") + "] " + message + "\n";
}

// Master initialization function
function initializeWorkspace() {
   // Create log note first
   create("/Log");
   $Badge("/Log") = "edit";
   $Text("/Log") = "";
   
   logMessage("Starting workspace initialization");
   
   logMessage("Calling finishHints()...");
   finishHints();
   
   logMessage("Calling instantiateTemplates()...");
   instantiateTemplates();
   
   logMessage("Calling instantiatePrototypes()...");
   instantiatePrototypes();
   
   logMessage("Calling instantiateProjects()...");
   instantiateProjects();
   
   logMessage("Calling instantiateResources()...");
   instantiateResources();
   
   logMessage("Workspace initialization completed successfully");
   logMessage("=== SUMMARY ===");
   logMessage("Check if /Prototypes exists: " + exists("/Prototypes"));
   logMessage("Check if /Projects exists: " + exists("/Projects"));  
   logMessage("Check if /Templates exists: " + exists("/Templates"));
   logMessage("Check if /Resources exists: " + exists("/Resources"));
   logMessage("Check if /Hints exists: " + exists("/Hints"));
   return true;
}
