create("/Log");

create("/Hints");

$Text("/Log") = $Text("/Log") + "Created Library\n";

create("/Hints/Preview");
$Badge("/Hints/Preview") = "display-chart";  
$Text("/Log") = $Text("/Log") + "Created Preview\n";

create("/Projects");
$Badge("/Projects") = "folder2";
$Text("/Log") = $Text("/Log") + "Created Projects\n";

create("/Resources");

// Create a simple utility module

$Text("/Log") = $Text("/Log") + "Created Utils module\n";
