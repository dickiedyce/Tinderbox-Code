// Prototype Builder for Tinderbox Workspace
// Functions for creating and configuring note prototypes

// Function: createPrototypesContainer
// Purpose: Create the main /Prototypes folder with proper configuration
function createPrototypesContainer() {
   logMessage("Creating /Prototypes folder...");
   
   create("/Prototypes");
   $Badge("/Prototypes") = "file-cabinet";
   $OnAdd("/Prototypes") = '$IsPrototype=true;$DisplayedAttributes="OnAdd;Badge;HTMLExportTemplate"';
   $DisplayedAttributes("/Prototypes") = "OnAdd";
   
   logMessage("/Prototypes folder created");
}

// Function: createActionPrototype
// Purpose: Create prototype for action notes
function createActionPrototype() {
   create("/Prototypes","Action");
   $Badge("/Prototypes/Action") = "crosshair";
   $Color("/Prototypes/Action") = "lighter black";
   $IsAction("/Prototypes/Action") = true;
   $SmartQuotes("/Prototypes/Action") = false;
   $DisplayedAttributes("/Prototypes/Action") = "OnAdd";
}

// Function: createFolderPrototype  
// Purpose: Create prototype for folder notes
function createFolderPrototype() {
   create("/Prototypes","pFolder");
   $Badge("/Prototypes/pFolder") = "folder2";
   $Color("/Prototypes/pFolder") = "lighter black";
   $IsPrototype("/Prototypes/pFolder") = true;
   $SmartQuotes("/Prototypes/pFolder") = false;
   $DisplayedAttributes("/Prototypes/pFolder") = getAttributeListForPrototype("folder");
   $HTMLExportTemplate("/Prototypes/pFolder") = "/Templates/Preview Folder";
}

// Function: instantiateAllPrototypes
// Purpose: Create all prototypes in the correct order
function instantiateAllPrototypes() {
   logMessage("Creating all prototypes...");
   
   createPrototypesContainer();
   createActionPrototype();
   createFolderPrototype();
   createHTMLPrototype();
   createResourcePrototype();
   createRolePrototype();
   createStoryPrototype();
   createProjectPrototype();
   createFRPrototype();
   
   logMessage("All prototypes created successfully");
}

// Function: validatePrototypes
// Purpose: Verify all prototypes were created correctly
function validatePrototypes() {
   var(requiredPrototypes, missingPrototypes){
      requiredPrototypes = "Action;pFolder;HTML;pResource;pRole;pStory;pProject;pFR";
      missingPrototypes = "";
      
      requiredPrototypes.split(";").each(prototype){
         if(!exists("/Prototypes/" + prototype)){
            missingPrototypes = missingPrototypes + prototype + ";";
         }
      }
      
      if(missingPrototypes != "") {
         logMessage("WARNING: Missing prototypes: " + missingPrototypes);
         return false;
      } else {
         logMessage("All prototypes validated successfully");
         return true;
      }
   }
}
